## 버그 원인: Stale Closure 문제

### 핵심 문제

`MenuContext.tsx`에서 **JavaScript의 stale closure 문제**가 발생했습니다.

#### 문제가 있던 코드 패턴:

```typescript
// fetchMenus 콜백이 "로그인 전" 상태를 캡처
const fetchMenus = useCallback(async () => {
  if (!isAuthenticated || !user) {  // ← 이 값이 false/null로 고정됨!
    return;  // 즉시 반환되어 API 호출 안됨
  }
  // API 호출...
}, [isAuthenticated, user]);

// useEffect에서 콜백 호출
useEffect(() => {
  if (isAuthenticated && user) {
    fetchMenus();  // ← stale closure 값 사용
  }
}, [isAuthenticated, user?.id]);
```

### 왜 이런 문제가 발생하는가?

| 시점 | isAuthenticated | user | fetchMenus가 캡처한 값 |
|------|-----------------|------|------------------------|
| 로그인 전 | false | null | false / null |
| 로그인 완료 | **true** | **{...}** | false / null (이전 값!) |
| fetchMenus 실행 | - | - | **첫 번째 if문에서 return!** |

### 왜 재로그인은 작동하는가?

1. 로그아웃 시 `MenuProvider`가 언마운트됨
2. 로그인 페이지에서 대기
3. 재로그인 시 `MenuProvider`가 새로 마운트 → **새 클로저 생성**
4. 새 콜백은 최신 값을 캡처하므로 정상 작동

  await oracledb.createPool({
    user: process.env.DB_USER || 'corenext',
    password: <REDACTED> || '<REDACTED_PASSWORD>',

    poolMax: 10,
    poolIncrement: 1
  });
  console.log('Connection pool 생성됨');
}

async function query(sql, params = []) {
  let connection;
  try {
    connection = await oracledb.getConnection();
    const result = await connection.execute(sql, params, { outFormat: oracledb.OUT_FORMAT_OBJECT });
    return result.rows;
  } finally {
    if (connection) await connection.close();
  }
}

module.exports = { init, query };

### 3. Node.js 기반 간단한 CLI 만들기

node-oracledb로 간단한 CLI 래퍼를 만들 수 있습니다.

```bash
# 설치
npm install oracledb readline
```

```javascript
// oracle-cli.js
const oracledb = require('oracledb');
const readline = require('readline');

const config = {
  user: process.env.DB_USER || 'corenext',
  password: <REDACTED> || '<REDACTED_PASSWORD>',
  connectString: process.env.DB_CONNECT_STRING || 'localhost:1521/ORCL'
};

async function main() {
  const conn = await oracledb.getConnection(config);
  console.log('Oracle 연결됨. SQL을 입력하세요. (exit로 종료)\n');

  const rl = readline.createInterface({
    input: process.stdin,
    output: process.stdout,
    prompt: 'SQL> '
  });

  rl.prompt();

  rl.on('line', async (line) => {
    const sql = line.trim();
    if (sql.toLowerCase() === 'exit') {
      await conn.close();
      process.exit(0);
    }
    if (!sql) { rl.prompt(); return; }

    try {
      const result = await conn.execute(sql, [], { outFormat: oracledb.OUT_FORMAT_OBJECT });
      if (result.rows) {
        console.table(result.rows);
        console.log(`${result.rows.length} rows\n`);
      } else {
        console.log('OK\n');
      }
    } catch (err) {
      console.error('Error:', err.message, '\n');
    }
    rl.prompt();
  });
}

main().catch(console.error);
```
