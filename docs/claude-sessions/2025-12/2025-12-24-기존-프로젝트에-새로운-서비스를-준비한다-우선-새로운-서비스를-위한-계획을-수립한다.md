### 2. 데이터베이스 스키마 설계

```sql
-- 체크시트 템플릿 (검사 양식 정의)
CREATE TABLE checksheet_templates (
    id UUID PRIMARY KEY,
    code VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(200) NOT NULL,
    description TEXT,
    category VARCHAR(100),
    version INTEGER DEFAULT 1,
    status VARCHAR(20) DEFAULT 'active',  -- active, inactive, archived
    created_by UUID REFERENCES users(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 체크시트 항목 (검사 항목 정의)
CREATE TABLE checksheet_items (
    id UUID PRIMARY KEY,
    template_id UUID REFERENCES checksheet_templates(id) ON DELETE CASCADE,
    parent_id UUID REFERENCES checksheet_items(id),  -- 계층 구조 지원
    sort_order INTEGER NOT NULL,
    item_code VARCHAR(50),
    item_name VARCHAR(200) NOT NULL,
    item_type VARCHAR(30) NOT NULL,  -- checkbox, text, number, select, photo, signature
    options JSONB,  -- select 옵션, 범위 제한 등
    required BOOLEAN DEFAULT false,
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 검사 실시 기록
CREATE TABLE inspections (
    id UUID PRIMARY KEY,
    template_id UUID REFERENCES checksheet_templates(id),
    inspection_code VARCHAR(50) UNIQUE NOT NULL,
    target_name VARCHAR(200),  -- 검사 대상
    target_id VARCHAR(100),    -- 외부 시스템 참조 ID
    inspector_id UUID REFERENCES users(id),
    status VARCHAR(20) DEFAULT 'draft',  -- draft, in_progress, completed, submitted
    started_at TIMESTAMP,
    completed_at TIMESTAMP,
    submitted_at TIMESTAMP,
    location VARCHAR(200),
    notes TEXT,
    sync_status VARCHAR(20) DEFAULT 'synced',  -- synced, pending, conflict
    client_id VARCHAR(100),  -- 모바일 기기 식별자
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 검사 결과 (각 항목별 결과)
CREATE TABLE inspection_results (
    id UUID PRIMARY KEY,
    inspection_id UUID REFERENCES inspections(id) ON DELETE CASCADE,
    item_id UUID REFERENCES checksheet_items(id),
    value TEXT,  -- 입력된 값
    value_type VARCHAR(30),  -- text, number, boolean, json
    is_passed BOOLEAN,  -- 합격/불합격
    remarks TEXT,  -- 비고
    photo_urls JSONB,  -- 첨부 사진
    recorded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    offline_created_at TIMESTAMP,  -- 오프라인에서 생성된 시간
    sync_version INTEGER DEFAULT 1
);

-- 오프라인 동기화 큐
CREATE TABLE sync_queue (
    id UUID PRIMARY KEY,
    client_id VARCHAR(100) NOT NULL,
    entity_type VARCHAR(50) NOT NULL,  -- inspection, inspection_result
    entity_id UUID NOT NULL,
    operation VARCHAR(20) NOT NULL,  -- create, update, delete
    payload JSONB NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    synced_at TIMESTAMP,
    sync_status VARCHAR(20) DEFAULT 'pending',  -- pending, synced, failed
    error_message TEXT,
    retry_count INTEGER DEFAULT 0
);
```

#### 6.3 동기화 프로세스

```typescript
// 1. 사전 다운로드 (온라인 상태에서)
async function downloadForOffline(templateIds: string[]) {
  // 선택한 템플릿과 항목 다운로드
  const data = await inspectionApi.get('/sync/download', { templateIds });
  await offlineStorage.saveTemplates(data.templates);
  await offlineStorage.updateMeta({ lastSyncAt: new Date() });
}

// 2. 오프라인 검사 저장
async function saveOfflineInspection(inspection: InspectionData) {
  await offlineStorage.saveInspection({
    ...inspection,
    status: 'pending_sync',
    lastModified: new Date()
  });
  
  // 동기화 큐에 추가
  await offlineStorage.addToSyncQueue({
    operation: 'create',
    entityType: 'inspection',
    payload: inspection
  });
}

// 3. 네트워크 복구 시 동기화
async function syncPendingData() {
  const queue = await offlineStorage.getSyncQueue();
  
  for (const item of queue) {
    try {
      await inspectionApi.post('/sync/upload', item.payload);
      await offlineStorage.removeFromQueue(item.id);
      await offlineStorage.updateInspectionStatus(item.payload.id, 'synced');
    } catch (error) {
      if (error.status === 409) {
        // 충돌 발생 - 사용자에게 알림
        await handleConflict(item);
      } else {
        // 재시도 카운터 증가
        await offlineStorage.incrementRetry(item.id);
      }
    }
  }
}
```

        </Button>
